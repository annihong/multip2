Expected Jaccard index under random edge permutation, given \(m_s\) edges in the first network, \(m_t\) edges in the second, and \(N\) total possible edges.

# Expected Jaccard index of two networks under random edge permutation.
# ds: density of the first network
# dt: density of the second network
# n: number of nodes in the network
expected_jaccard <- function(ds, dt, n) {
  N <- n * (n - 1)  # total number of possible edges for a directed network
  ms <- ds * N  # number of edges in the first network
  mt <- dt * N  # number of edges in the second network
  
  k_min <- max(0, ms + mt - N)
  k_max <- min(ms, mt)
  expected_jaccard <- 0
  for (k in k_min:k_max) {
    val <- k / (ms + mt - k)
    # numer <- choose(ms, k) * choose(N - ms, mt - k)
    # demo <-  choose(N, mt)
    # prob <- numer / demo
    prob <- dhyper(k, ms, N - ms, mt)
    expected_jaccard <- expected_jaccard + val * prob
  }
  return(expected_jaccard) 
}
